## Code Blocks

Standard markdown code blocks work great with syntax highlighting:

```typescript
// TypeScript example with syntax highlighting
interface User {
  id: string;
  name: string;
  email: string;
}

const createUser = (data: Partial<User>): User => {
  return {
    id: crypto.randomUUID(),
    name: data.name ?? 'Anonymous',
    email: data.email ?? 'noreply@ocean-log.dev',
  };
};
```

You can also use the CodeBlock component for enhanced styling:

<CodeBlock title="config.ts" language="typescript">
```typescript
export const config = {
  siteName: 'ocean-log',
  siteUrl: 'https://ocean-log.dev',
  author: 'Developer',
};
```
</CodeBlock>
